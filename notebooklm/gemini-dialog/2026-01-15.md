```
import { z } from "zod";
import { ScreenDef, ScreenSchema } from "./schemas";

// 1. Initialize Gemini
// Ensure GEMINI_API_KEY is set in your environment variables
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY || "");

// 2. Define the Schema for the Model
// We map our Zod schema to the Gemini SDK's expected format.
// This ensures the model *cannot* hallucinations invalid fields.
const modelSchema = {
  description: "A definition of a server-driven UI screen.",
  type: SchemaType.OBJECT,
  properties: {
    screenTitle: { type: SchemaType.STRING, description: "Title of the dashboard/page" },
    sections: {
      type: SchemaType.ARRAY,
      items: {
        type: SchemaType.OBJECT,
        properties: {
          id: { type: SchemaType.STRING },
          title: { type: SchemaType.STRING },
          description: { type: SchemaType.STRING },
          layout: { type: SchemaType.STRING, enum: ["single", "two-column", "three-column", "grid"] },
          children: {
            type: SchemaType.ARRAY,
            items: {
              type: SchemaType.OBJECT,
              properties: {
                type: { type: SchemaType.STRING, enum: ["StatCard", "DataGrid", "Markdown", "ActionPanel"] },
                props: { type: SchemaType.OBJECT, nullable: true } 
                // Note: In a full prod app, we would recursively define props schema here.
                // For this POC, we trust the model to match the prop names based on the system prompt descriptions.
              }
            }
          }
        },
        required: ["id", "layout", "children"]
      }
    }
  },
  required: ["screenTitle", "sections"]
};

// 3. The System Prompt (The Persona)
const SYSTEM_PROMPT = `
You are an expert UI Designer and Data Analyst.
Your job is to generate a JSON definition for a dashboard based on the user's request.

You have access to the following component library:

1. StatCard: For single metrics. Props: { title, value, trend (number), trendDirection ('up'|'down'|'neutral') }
2. DataGrid: For tables. Props: { title, columns (string[]), rows (object[]) }
3. ActionPanel: For suggested user actions. Props: { prompt, buttons: [{ label, endpoint, variant }] }
   - IMPORTANT: For 'endpoint', invent a plausible API path (e.g., '/api/actions/export').
4. Markdown: For text explanations. Props: { content }

RULES:
- Be concise.
- Choose the best layout for the data.
- If the user asks for analysis, perform it and display the result in a Markdown component or StatCard.
`;

// 4. The Generator Function
export async function generateScreen(userPrompt: string): Promise<ScreenDef> {
  const model = genAI.getGenerativeModel({
    model: "gemini-1.5-flash",
    generationConfig: {
      responseMimeType: "application/json",
      responseSchema: modelSchema,
    },
    systemInstruction: SYSTEM_PROMPT,
  });

  try {
    const result = await model.generateContent(userPrompt);
    const json = JSON.parse(result.response.text());
    
    // Optional: Validate with Zod before returning (Double Defense)
    // const validated = ScreenSchema.parse(json); 
    
    return json as ScreenDef;
  } catch (error) {
    console.error("Gemini Generation Error:", error);
    // Fallback UI in case of failure
    return {
      screenTitle: "System Error",
      sections: [{
        id: "error",
        layout: "single",
        children: [{ type: "Markdown", props: { content: `**Error generating UI:** ${error.message}` } }]
      }]
    };
  }
}
```

# Project Brief: Project REIFY
**"Turning Abstract Intent into Concrete Interface"**


### 1. The Core Concept (Concept-03)
**Project Reify** is a Server-Driven Generative UI system. Unlike standard GenUI (where the AI writes volatile code), Reify uses the AI as a **Selector**. The AI maps user intent to a pre-defined, type-safe library of server-side components (The "Things").


**The Stack:**


- **Substrate:** Bun (Runtime)
- **Server:** Hono (Edge-ready HTTP)
- **Templating:** JSX (Type-safe component definitions)
- **The Glue:** DataStar v1 (Hypermedia + Fine-grained Reactivity)

### 2. Operational Architecture (OH-040: Factored Design)
We separate the system into three distinct distinct "Organs":


1. **The Component Registry (The Territory)**


- A rigid collection of Hono/JSX components (e.g., `<DataGrid />`, `<MetricCard />`, `<ChatBubble />`).
- Each component exports a Zod schema defining its props.
- *Role:* Pure rendering. No logic.
2. **The Reasoning Engine (The Mentation)**


- The LLM (Gemini) receives the user prompt + current state.
- It outputs a **JSON Display Tree** (not code).
- *Example Output:* `{"layout": "split_view", "left": {"id": "chat"}, "right": {"component": "DataGrid", "props": {...}}}`.
3. **The Hydration Pump (The DataStar Bridge)**


- Hono receives the JSON from the Engine.
- It maps the JSON to the JSX Registry.
- It streams the rendered HTML fragments to the client via Server-Sent Events (SSE), utilizing DataStar's `data-on-load` or specific signal triggers.

### 3. Handling User Reactivity
Since we are avoiding client-side framework bloat, we handle reactivity via the **DataStar Signal Protocol**. Here are the three implementation tiers:


**Option A: Ephemeral Reactivity (The "Skin" Layer)**


- *Use Case:* Toggling visibility, tabs, opening modals, simple arithmetic.
- *Method:* Use DataStar's `data-store`.
- *Mechanism:* The state lives in the DOM. No server roundtrip.

```
<div data-store="{ isOpen: false }">
  <button data-on-click="$isOpen = !$isOpen">Toggle Details</button>
  <div data-show="$isOpen">Secret Content</div>
</div>

```


**Option B: The RPC Reflex (The "Spine" Layer)**


- *Use Case:* Sorting a list, filtering data, validating a form field.
- *Method:* DataStar Action to a Hono route.
- *Mechanism:* User clicks -> `data-fetch` hits Hono -> Hono re-renders *only* that component -> Response merges HTML fragment back into the DOM.

*Latency:* 

**Option C: The Agentic Loop (The "Brain" Layer)**


- *Use Case:* "Analyze this data," "Change the layout," "Zoom into this graph."
- *Method:* The user input is sent as a prompt to the Reasoning Engine.
- *Mechanism:*

1. User types "Show me sales by region" (DataStar sends input to server).
- Server invokes Agent.
- Agent generates new **Display Tree JSON**.
- Server renders the *entire* new UI layout.
- DataStar swaps the `<body>` or main container with the new content via SSE.

**Step 1: The Registry.** Build 3 "dumb" Hono/JSX components (Card, List, Chart). Ensure they render individually.
2. **Step 2: The Schema.** Define the Zod schemas for those 3 components.
3. **Step 3: The Mock Brain.** Create a Hono route that ignores the LLM and hardcodes a return of the JSON Display Tree to prove the rendering pipeline works with DataStar.
4. **Step 4: The Integration.** Connect the LLM to generate the JSON based on a string input.

### Why "DataStar"?
It allows us to treat the UI as a **State Machine driven by the Server**. The Agent owns the state; the browser just reflects it. This eliminates the "State Synchronization Gap" that plagues React/Next.js GenUI implementations.

### The Analysis: "Gemini GenUI" vs. "Hono/SSR/JSX"
The blog post you read likely describes the "Google GenUI" pattern: the model acts as a frontend developer, writing raw React/HTML/CSS code on the fly.


- **Pros:** Infinite flexibility.
- **Cons:** High entropy. The model can hallucinate invalid props, break layouts, or inject insecure code. It requires a heavy "sleeve" of prompt engineering to keep it safe.

**Your proposal (Hono/SSR/JSX) is superior because it separates *Intent* from *Implementation*.**


Instead of asking the AI to *write* the UI code (high risk), we ask it to *select and configure* the UI from a trusted library (low risk).


### The Proposed Architecture: "The Component Registry"
We can build this using a **Factored Design** approach.


#### 1. The Territory (The Components)
We build a set of robust, pre-tested Hono/JSX components. These are our "Things". They live on the server.


- `StandardCard({ title, body, footer })`
- `DataGrid({ columns, rows })`
- `ActionPanel({ prompt, buttonText })`

#### 2. The Map (The Schema)
We define a strict JSON schema (Zod) that represents the valid inputs for these components. The AI does not generate JSX; it generates **structured JSON** that maps to these components.


#### 3. The Mentation Bridge (The Agent)
The Agent receives a user query (e.g., "Analyze this CSV"), performs the logic, and then outputs a **Display Tree** JSON:

```
  "layout": "dashboard",

    {
      "type": "ActionPanel",
      "props": { "label": "Filter", "actionId": "filter_rows" }
    }
  ]
}

```

#### 4. The Renderer (Hono)
A simple Hono route takes this JSON, looks up the components in the registry, and server-side renders them into high-speed HTML.

1. **Type Safety (OH-084):** We can validate the AI's output against the Zod schema *before* rendering. If the AI hallucinates a non-existent prop, we catch it in the logic layer, not the view layer.

3. **Low-Tech Defence (Principle of Effective Low-Tech Defence):** We avoid the complexity of managing client-side state synchronization. If we need interactivity, we can use **HTMX** to swap out components on the server, keeping the client "dumb" and the Agent in control.
