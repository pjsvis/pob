### 2. The Orchestration Script (`bin/amalfa-forge`)
This Bash script utilizes `gum` to create an interactive "Augment-style" interface. It handles the transition from selection to manifest creation.


```
#!/bin/bash
# bin/amalfa-forge: The Phase 1 Mission Forge

# Ensure we have a local lexicon for grounding
LEXICON_FILE="$HOME/.amalfa/intel/conceptual-lexicon.json"

# 1. SELECT BRIEF (The Target 'Thing')
BRIEF=$(find briefs -name "*.md" | gum filter --header "1. SELECT PRIMARY BRIEF")
[ -z "$BRIEF" ] && exit 0

# 2. GENERATE GLOBAL ID (GitHub Source of Record)
TITLE=$(basename "$BRIEF" .md | tr '-' ' ')
echo "Forging Remote Identity..." | gum style --foreground 212
GH_URL=$(gh issue create --title "Mission: $TITLE" --body "Initial Brief Reference: $BRIEF" --label "agent-task")
ISSUE_ID=$(echo "$GH_URL" | grep -oE "[0-9]+$")

# 3. SELECT HEURISTICS (Playbooks)
PLAYBOOKS=$(find playbooks -name "*.md" | gum choose --no-limit --header "2. ATTACH OPERATIONAL PLAYBOOKS")

# 4. CAPTURE RAW INSTRUCTIONS (Additional Stuff)
EXTRA_STUFF=$(gum write --placeholder "3. ADDITIONAL CONTEXT OR INSTRUCTIONS...")

# 5. ASSEMBLE THE MANIFEST (The RHS Context)
# We use jq to build a valid JSON artifact
jq -n \
  --arg id "$ISSUE_ID" \
  --arg url "$GH_URL" \
  --arg brief "$BRIEF" \
  --arg pbs "$PLAYBOOKS" \
  --arg instr "$EXTRA_STUFF" \
  '{
    mission_id: $id,
    github_url: $url,
    brief_path: $brief,
    playbooks: ($pbs | split("\n")),
    raw_instructions: $instr,
    status: "forged",
    timestamp: (now | strftime("%Y-%m-%dT%H:%M:%SZ"))
  }' > .current_mission.json

# 6. LOCAL STATE LOCK (td integration)
td create "Mission #$ISSUE_ID: $TITLE" --id "$ISSUE_ID"
td link "$ISSUE_ID" "$BRIEF"
for PB in $PLAYBOOKS; do td link "$ISSUE_ID" "$PB"; done

# 7. FINAL MENTATION CHECK (Persona Grounding)
gum style --border double --margin "1 1" --padding "1 2" --foreground 212 \
    "MISSION #$ISSUE_ID FORGED" \
    "LHS (Sources): $(echo "$PLAYBOOKS" | wc -l) Playbooks + 1 Brief" \
    "RHS (Artifact): .current_mission.json created"

ACTION=$(gum choose "KICKOFF (Engage Agent)" "STASH (Backlog)")

if [ "$ACTION" == "KICKOFF (Engage Agent)" ]; then
    td start "$ISSUE_ID"
    # Update manifest for Phase 2: Mentation
    jq '.status = "active"' .current_mission.json > .tmp && mv .tmp .current_mission.json
    gum style --foreground 120 "Agent Substrate engaged for #$ISSUE_ID. Check GitHub for updates."
else
    gum style --foreground 212 "Mission #$ISSUE_ID stashed in backlog."
fi

```
