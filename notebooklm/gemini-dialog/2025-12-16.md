 * Extracts sections from Markdown into a structured Bento Box array.
 * Uses AST to reveal inherent structure without AI intervention.
 */
export function extractBentoBoxes(markdown: string): BentoBox[] {
  const tree = fromMarkdown(markdown);
  const boxes: BentoBox[] = [];
  let currentBox: Partial<BentoBox> | null = null;
  let contentBuffer: string[] = [];

  // Iterate through AST nodes to identify structural boundaries
  for (const node of tree.children) {
    if (node.type === 'heading') {
      // If we were already building a box, finalize it
      if (currentBox) {
        currentBox.content = contentBuffer.join('\n').trim();
        boxes.push(currentBox as BentoBox);
      }

      // Start a new Bento Box
      const headerNode = node as Heading;
      const title = (headerNode.children[0] as any)?.value || 'Untitled';
      
      currentBox = {
        id: `box-${Math.random().toString(36).substr(2, 9)}`,
        level: headerNode.depth,
        title: title,
      };
      contentBuffer = [];
    } else {
      // Append non-heading nodes to the current box's content buffer
      if (currentBox) {
        // Here we could use mdast-util-to-markdown to stringify individual nodes
        contentBuffer.push((node as any).value || ''); 
      }
    }
  }

  // Flush the final box
  if (currentBox) {
    currentBox.content = contentBuffer.join('\n').trim();
    boxes.push(currentBox as BentoBox);
  }

  return boxes;
}
